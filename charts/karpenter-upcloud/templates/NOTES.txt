3. Create an UpCloudNodeClass and NodePool to start provisioning:

   kubectl apply -f - <<EOF
   apiVersion: karpenter.upcloud.com/v1alpha1
   kind: UpCloudNodeClass
   metadata:
     name: default
   spec:
     zone: de-fra1
     plan: CLOUDNATIVE-2xCPU-4GB
     storageGB: 20
     storageTier: maxiops
     sshKeys:
       - "ssh-ed25519 AAAA..."
   ---
   apiVersion: karpenter.sh/v1
   kind: NodePool
   metadata:
     name: default
   spec:
     template:
       spec:
         nodeClassRef:
           group: karpenter.upcloud.com
           kind: UpCloudNodeClass
           name: default
     disruption:
       consolidationPolicy: Never
   EOF
